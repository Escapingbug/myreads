cmake_minimum_required(VERSION 3.22)
project(zijian_tokenizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/sentencepiece EXCLUDE_FROM_ALL)
add_library(zijian_tokenizer SHARED tokenizer_jni.cpp)
target_include_directories(zijian_tokenizer PRIVATE vendor/sentencepiece/src)
target_link_libraries(zijian_tokenizer sentencepiece-static log)
