cmake_minimum_required(VERSION 3.22)
project(zijian_tokenizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/sentencepiece EXCLUDE_FROM_ALL)
add_subdirectory(vendor/openfst EXCLUDE_FROM_ALL)
add_library(zijian_tokenizer SHARED tokenizer_jni.cpp wetext_jni.cpp)
target_include_directories(zijian_tokenizer PRIVATE vendor/sentencepiece/src)
target_link_libraries(zijian_tokenizer sentencepiece-static official_fst)
if(ANDROID)
    target_link_libraries(zijian_tokenizer log)
    # Android 15 supports 16 KB memory pages.
    target_link_options(zijian_tokenizer PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_package(JNI REQUIRED)
    target_include_directories(zijian_tokenizer PRIVATE ${JNI_INCLUDE_DIRS})
endif()
add_subdirectory(bertvits)
